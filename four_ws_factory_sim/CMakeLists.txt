cmake_minimum_required(VERSION 3.8)
project(four_ws_factory_sim)

find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(gazebo REQUIRED)
find_package(gazebo_ros REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(tf2_ros REQUIRED)

add_library(factory_vehicle_control SHARED src/vehicle_control.cpp)
target_compile_features(factory_vehicle_control PRIVATE cxx_std_17)
target_include_directories(factory_vehicle_control PRIVATE include ${GAZEBO_INCLUDE_DIRS})
target_link_libraries(factory_vehicle_control ${GAZEBO_LIBRARIES})
ament_target_dependencies(factory_vehicle_control gazebo_ros rclcpp geometry_msgs nav_msgs sensor_msgs tf2_ros)
add_library(factory_mid360_pattern SHARED src/mid360_pattern.cpp)
target_compile_features(factory_mid360_pattern PRIVATE cxx_std_17)
target_include_directories(factory_mid360_pattern PRIVATE include ${GAZEBO_INCLUDE_DIRS})
target_link_libraries(factory_mid360_pattern ${GAZEBO_LIBRARIES})
ament_target_dependencies(factory_mid360_pattern gazebo_ros rclcpp sensor_msgs)

ament_python_install_package(${PROJECT_NAME})
install(TARGETS factory_vehicle_control factory_mid360_pattern LIBRARY DESTINATION lib)
install(DIRECTORY launch worlds meshes models DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/pallet_detector scripts/pick_pallet scripts/check_pick DESTINATION lib/${PROJECT_NAME})
ament_package()
